## DATABASE SCHEMA ADDITIONS (Supabase)

```sql
-- Loans Table
CREATE TABLE loans (
  id UUID PRIMARY KEY,
  driver_id UUID REFERENCES drivers(id),
  amount_requested INT,
  down_payment INT,
  monthly_payment INT,
  status TEXT DEFAULT 'PENDING', -- PENDING, APPROVED, ACTIVE, PAID_OFF
  auto_deduct BOOLEAN DEFAULT false,
  created_at TIMESTAMP,
  approved_at TIMESTAMP,
  van_title TEXT,
  next_payment_due DATE
);

-- Jobs Table
CREATE TABLE jobs (
  id UUID PRIMARY KEY,
  owner_id UUID REFERENCES drivers(id),
  route_id UUID REFERENCES routes(id),
  title TEXT,
  pay_per_month INT,
  status TEXT DEFAULT 'OPEN', -- OPEN, HIRED, FILLED
  hired_driver_id UUID REFERENCES drivers(id),
  created_at TIMESTAMP,
  filled_at TIMESTAMP
);

-- Success Stories Table
CREATE TABLE success_stories (
  id UUID PRIMARY KEY,
  driver_id UUID REFERENCES drivers(id),
  title TEXT,
  content TEXT,
  metrics JSONB, -- {started_families: 8, current_families: 20, ...}
  published BOOLEAN DEFAULT false,
  published_at TIMESTAMP,
  created_at TIMESTAMP
);

-- Alter drivers table
ALTER TABLE drivers ADD COLUMN tier TEXT DEFAULT 'DRIVER'; -- DRIVER, OWNER, FLEET
ALTER TABLE drivers ADD COLUMN earnings_this_month INT DEFAULT 0;
ALTER TABLE drivers ADD COLUMN num_routes INT DEFAULT 0;
ALTER TABLE drivers ADD COLUMN hired_drivers INT DEFAULT 0;
```
